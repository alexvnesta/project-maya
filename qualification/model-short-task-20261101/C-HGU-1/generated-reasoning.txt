import unittest

def _valid_endpoint(value):
    return isinstance(value, int) and not isinstance(value, bool)

def coalesce_ranges(ranges):
    normalized = []
    for index, item in enumerate(ranges):
        if not isinstance(item, (list, tuple)):
            raise TypeError("range at index {} is not a pair".format(index))
        if len(item) != 2:
            raise TypeError("range at index {} must contain exactly two endpoints".format(index))
        start, end = item
        if not _valid_endpoint(start):
            raise TypeError("start endpoint at index {} must be an integer and not bool".format(index))
        if not _valid_endpoint(end):
            raise TypeError("end endpoint at index {} must be an integer and not bool".format(index))
        if start > end:
            raise ValueError("start endpoint at index {} must not exceed end".format(index))
        if start != end:
            normalized.append((start, end))
    normalized.sort()
    merged = []
    for start, end in normalized:
        if not merged or start > merged[-1][1]:
            merged.append((start, end))
        else:
            last_start, last_end = merged[-1]
            if end > last_end:
                merged[-1] = (last_start, end)
    return merged
