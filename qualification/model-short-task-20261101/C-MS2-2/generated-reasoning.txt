        if not isinstance(pair, (list, tuple)) or len(pair) != 2:

        if _is_int(start) etc:

    intervals.sort(key=lambda x: (x[0], x[1]))
    merged = []
    for start, end in intervals:
        if not merged:
            merged.append((start, end))

            prev_start, prev_end = merged[-1]
