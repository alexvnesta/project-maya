def coalesce_ranges(ranges):
    normalized = []
    for item in ranges:
        try:
            start, end = item
        except (TypeError, ValueError):
            raise TypeError("each range must be a pair of endpoints") from None
        if not _is_int(start) or not _is_int(end):
            raise TypeError("range endpoints must be integers, not bool")
        if start > end:
            raise ValueError("range start must not be greater than end")
        if start != end:
            normalized.append((start, end))
    if not normalized:
        return []
    normalized.sort()
    merged = []
    for start, end in normalized:
        if not merged:
            merged.append((start, end))
        else:
            last_start, last_end = merged[-1]
            if start <= last_end:
                if end > last_end:
                    merged[-1] = (last_start, end)
            else:
                merged.append((start, end))
    return merged

```
import unittest

def _is_integer(value):
    return isinstance(value, int) and not isinstance(value, bool)

def coalesce_ranges(ranges):
    """..."""
    intervals = []
    for item in ranges:
        try:
            start, end = item
        except (TypeError, ValueError):
            raise TypeError("each range must be a pair of endpoints") from None

        if not _is_integer(start) or not _is_integer(end):
            raise TypeError("range endpoints must be integers and not bool")
        if start > end:
            raise ValueError("range start must not exceed end")
        if start == end:
            continue
        intervals.append((start, end))

    if not intervals:
        return []

    intervals.sort()
    merged = []
    for start, end in intervals:
        if not merged:
            merged.append((start, end))
            continue
        last_start, last_end = merged[-1]
        if start <= last_end:
            if end > last_end:
                merged[-1] = (last_start, end)
        else:
            merged.append((start, end))
    return merged
```
