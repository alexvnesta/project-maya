import copy
import unittest

def _validate_endpoint(value):
    if isinstance(value, bool) or not isinstance(value, int):
        raise TypeError("range endpoints must be integers and not bool")
    return value

def coalesce_ranges(ranges):
    intervals = []
    for index, item in enumerate(ranges):
        try:
            start, end = item
        except (TypeError, ValueError):
            raise TypeError(f"range at index {index} must be a pair of endpoints") from None
        if isinstance(start, bool) or not isinstance(start, int):
            raise TypeError("range endpoints must be integers and not bool")
        if isinstance(end, bool) or not isinstance(end, int):
            raise TypeError("range endpoints must be integers and not bool")
        if start > end:
            raise ValueError("range start must be less than or equal to end")
        if start != end:
            intervals.append((start, end))
    intervals.sort(key=lambda interval: (interval[0], interval[1]))
    merged = []
    for start, end in intervals:
        if not merged:
            merged.append((start, end))
            continue
        previous_start, previous_end = merged[-1]
        if start <= previous_end:
            if end > previous_end:
                merged[-1] = (previous_start, end)
        else:
            merged.append((start, end))
    return merged
```

def coalesce_ranges(ranges):
    intervals = []
    for index, item in enumerate(ranges):
        try:
            if len(item) != 2:
                raise TypeError
            start = item[0]
            end = item[1]
        except (TypeError, ValueError, IndexError, KeyError):
            raise TypeError(f"range at index {index} must be a pair of endpoints") from None
        if isinstance(start, bool) or not isinstance(start, int):
            raise TypeError("range endpoints must be integers and not bool")
        if isinstance(end, bool) or not isinstance(end, int):
            raise TypeError("range endpoints must be integers and not bool")
        if start > end:
            raise ValueError("range start must not be greater than end")
        if start != end:
            intervals.append((start, end))
    intervals.sort(key=lambda pair: (pair[0], pair[1]))
    merged = []
    for start, end in intervals:
        if not merged:
            merged.append((start, end))
            continue
        previous_start, previous_end = merged[-1]
        if start <= previous_end:
            if end > previous_end:
                merged[-1] = (previous_start, end)
        else:
            merged.append((start, end))
    return merged
```

def coalesce_ranges(ranges):
    intervals = []
    for index, item in enumerate(ranges):
        try:
            if len(item) != 2:
                raise TypeError
            start = item[0]
            end = item[1]
        except (TypeError, ValueError, IndexError, KeyError):
            raise TypeError(
                f"range at index {index} must be a pair of integer endpoints"
            ) from None

        _require_integer_endpoint(start)
        _require_integer_endpoint(end)

        if start > end:
            raise ValueError("range start must not be greater than end")

        if start != end:
            intervals.append((start, end))

    intervals.sort(key=lambda interval: (interval[0], interval[1]))

    merged = []
    for start, end in intervals:
        if not merged:
            merged.append((start, end))
            continue

        previous_start, previous_end = merged[-1]
        if start <= previous_end:
            if end > previous_end:
                merged[-1] = (previous_start, end)
        else:
            merged.append((start, end))

    return merged
